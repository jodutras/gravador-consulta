// ===================================================================== // Worker de transcrição e resumo de consultas // Roda no VPS (Coolify), ao lado do Whisper e do Ollama. // // Ciclo: reivindica sessão -> baixa áudio -> transcreve -> resume -> // grava resultado -> APAGA o áudio (LGPD) // // Nenhum dado de paciente sai do servidor: Whisper e Ollama são locais. // ===================================================================== import { createClient } from '@supabase/supabase-js'; const { SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, WHISPER_URL, WHISPER_API_KEY, WHISPER_MODEL = 'deepdml/faster-whisper-large-v3-turbo-ct2', OLLAMA_URL = 'http://ollama:11434', OLLAMA_MODEL = 'qwen2.5:7b-instruct-q4_K_M', BUCKET = 'gravacoes', INTERVALO_MS = '10000', } = process.env; for (const [k, v] of Object.entries({ SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, WHISPER_URL, WHISPER_API_KEY, })) { if (!v) { console.error(`Variável obrigatória ausente: ${k}`); process.exit(1); } } const db = createClient(SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, { auth: { persistSession: false }, }); const log = (...a) => console.log(new Date().toISOString(), ...a); // --------------------------------------------------------------------- // Transcrição (Whisper self-hosted) // --------------------------------------------------------------------- async function transcrever(blob, nomeArquivo) { const form = new FormData(); form.append('file', blob, nomeArquivo); form.append('model', WHISPER_MODEL); form.append('language', 'pt'); form.append('response_format', 'verbose_json'); // O parâmetro prompt do Whisper tem janela curta (~224 tokens): lista de // medicamentos aqui NÃO funciona e já foi tentada. Quem confere nome de // remédio agora é o catálogo (conferir_medicamentos), depois da extração. // Aqui fica só a pista de contexto clínico geral. form.append('prompt', 'Consulta médica em português do Brasil. Termos clínicos, sintomas, ' + 'diagnósticos, medicamentos, dosagens em miligramas, pressão arterial, CID.'); const r = await fetch(`${WHISPER_URL}/v1/audio/transcriptions`, { method: 'POST', headers: { Authorization: `Bearer ${WHISPER_API_KEY}` }, body: form, signal: AbortSignal.timeout(45 * 60 * 1000), }); if (!r.ok) throw new Error(`Whisper ${r.status}: ${await r.text()}`); const j = await r.json(); return { texto: (j.text || '').trim(), duracao: j.duration ?? null }; } // --------------------------------------------------------------------- // Resumo estruturado (Ollama local) // --------------------------------------------------------------------- const INSTRUCOES = `Você é um escriba clínico. Recebe a transcrição bruta do áudio de uma consulta em português do Brasil e organiza o conteúdo nos campos do prontuário. Este é um RASCUNHO: o profissional de saúde vai conferir, corrigir e assinar. Ele não vai direto ao prontuário. SUA TAREFA PRINCIPAL É PREENCHER OS CAMPOS. Percorra a transcrição inteira e distribua TODO o conteúdo clínico entre os campos abaixo. Um campo só fica vazio quando a transcrição realmente não disser nada sobre aquele assunto. Deixar um campo vazio havendo informação correspondente é o pior erro possível — o rascunho fica inútil e o profissional precisa reescrever tudo do zero. Você DEVE reorganizar e resumir com suas palavras. Não é para copiar frases soltas nem esperar que o médico use os termos técnicos do campo. Se o paciente disse "minha pressão fica 14 por 9", isso é história da doença atual. Se disse "às vezes esqueço de tomar", isso é adesão ao tratamento e entra na história. Se o médico falou "não vou mexer na medicação, vou renovar a receita", isso é conduta. O que vai em cada campo: - queixa_principal: o motivo da consulta, em uma frase. Em retorno de doença crônica, é o acompanhamento dela. - historia_doenca_atual: evolução, sintomas, valores medidos, adesão ao tratamento, hábitos relevantes (alimentação, atividade, hidratação). - antecedentes: doenças prévias, cirurgias, internações, episódios anteriores, idade, alergias. - exame_fisico: apenas medidas e achados examinados na consulta. Se o profissional não examinou, deixe vazio. - hipoteses_diagnosticas: as hipóteses que o PROFISSIONAL verbalizou. Não crie diagnóstico próprio. - conduta: medicações mantidas ou alteradas, receitas, orientações dadas, exames pedidos, retorno agendado. - medicamentos_mencionados: nomes e dosagens citados. ATENÇÃO: a transcrição é automática e erra a grafia de nomes comerciais. Se o mesmo medicamento aparecer escrito de formas diferentes (ex.: "A-Block" e "Diabloque" são a mesma coisa mal transcrita), liste UMA vez só, com a grafia mais provável, e registre a dúvida em trechos_incertos. Nunca liste o mesmo remédio duas vezes. - observacoes: contexto relevante que não coube nos outros campos. Se a transcrição não for de uma consulta clínica, escreva isso aqui e deixe os demais campos vazios. - trechos_incertos: SOMENTE nomes de medicamento, dosagens ou números que soaram ambíguos na transcrição. Não use este campo para conteúdo que você simplesmente resolveu não classificar. O que NÃO fazer: - Não invente sintoma, medida, exame ou diagnóstico que não esteja na transcrição. - Não registre como exame físico algo que o paciente relatou em casa; isso é história. EXEMPLO (formato e nível de detalhe esperados): Transcrição: "Doutora, essa tosse não passa, já faz umas duas semanas. Seca, piora à noite. Febre não tive. Já tive bronquite quando era criança. Deixa eu escutar seu pulmão... murmúrio normal, sem chiado. Olha, parece alergia. Vou passar loratadina 10 miligramas, um por dia à noite. Se em uma semana não melhorar, o senhor volta." Resposta: {"queixa_principal":"Tosse seca há duas semanas.","historia_doenca_atual":"Tosse seca há cerca de duas semanas, com piora noturna. Nega febre.","antecedentes":"Bronquite na infância.","exame_fisico":"Ausculta pulmonar com murmúrio vesicular normal, sem sibilos.","hipoteses_diagnosticas":"Quadro sugestivo de origem alérgica, conforme avaliação do profissional.","conduta":"Prescrita loratadina 10 mg, um comprimido à noite. Retorno em uma semana caso não haja melhora.","medicamentos_mencionados":["Loratadina 10 mg"],"observacoes":"","trechos_incertos":[]} Responda SOMENTE com um objeto JSON válido nesse mesmo formato, sem nenhum texto fora dele.`; async function resumir(transcricao) { const r = await fetch(`${OLLAMA_URL}/api/chat`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ model: OLLAMA_MODEL, stream: false, format: 'json', // força saída JSON válida options: { temperature: 0.1, // extração, não criatividade num_ctx: 16384, // consulta longa gera transcrito grande }, messages: [ { role: 'system', content: INSTRUCOES }, { role: 'user', content: `Transcrição da consulta:\n\n${transcricao}` }, ], }), signal: AbortSignal.timeout(60 * 60 * 1000), // CPU é lenta; sem pressa }); if (!r.ok) throw new Error(`Ollama ${r.status}: ${await r.text()}`); const j = await r.json(); const texto = (j.message?.content || '').trim(); try { return JSON.parse(texto); } catch { return { erro_parse: true, texto_bruto: texto }; } } // --------------------------------------------------------------------- // Conferência dos medicamentos contra o catálogo // // POR QUE ISTO EXISTE // O Whisper erra a grafia de nome comercial. Num teste real ele escreveu // "A-Block" numa passagem e "Diabloque" na outra, para o mesmo remédio // (Ablok), e o modelo do resumo listou os dois como se fossem diferentes. // Colocar uma lista de medicamentos no prompt do Whisper foi tentado e // não funcionou: a janela do parâmetro é curta e existem dezenas de // milhares de medicamentos. // // A conferência NUNCA corrige em silêncio. Nome com boa correspondência // é anotado com o princípio ativo; dúvida e nome desconhecido vão para // "trechos_incertos", que a tela mostra marcado como CONFERIR. Trocar um // nome de remédio por conta própria seria a máquina adivinhando // prescrição — exatamente o que se quer evitar. // --------------------------------------------------------------------- async function conferirMedicamentos(resumo) { const originais = Array.isArray(resumo?.medicamentos_mencionados) ? resumo.medicamentos_mencionados.filter(m => typeof m === 'string' && m.trim()) : []; if (!originais.length) return resumo; const { data, error } = await db.rpc('conferir_medicamentos', { p_termos: originais }); if (error) { // catálogo indisponível não pode derrubar a sessão inteira: // o resumo segue com os nomes crus, que é o que já acontecia antes. log(`AVISO — conferência de medicamentos falhou: ${error.message}`); return resumo; } const conferidos = Array.isArray(data) ? data : []; const lista = []; const duvidas = Array.isArray(resumo.trechos_incertos) ? resumo.trechos_incertos.filter(Boolean).map(String) : []; for (const c of conferidos) { const termo = c?.termo || ''; if (c?.situacao === 'confirmado') { // mantém o termo original: é ele que carrega a dose e a posologia lista.push(c.principio ? `${termo} (${c.principio})` : termo); // nomes parecidos entre si merecem conferência mesmo quando o // primeiro bate bem: Losartana/Valsartana, Omeprazol/Esomeprazol if (Array.isArray(c.alternativas) && c.alternativas.length) { duvidas.push( `"${termo}": confira se não é ${c.alternativas.join(' ou ')} — nomes parecidos.` ); } } else if (c?.situacao === 'incerto') { lista.push(termo); duvidas.push( `"${termo}": possivelmente ${c.nome}` + (c.principio ? ` (${c.principio})` : '') + `. A transcrição pode ter errado o nome — confirmar.` ); } else { lista.push(termo); duvidas.push(`"${termo}": não encontrado no catálogo de medicamentos — confirmar.`); } } return { ...resumo, medicamentos_mencionados: lista, trechos_incertos: duvidas }; } // --------------------------------------------------------------------- // Processamento de uma sessão // --------------------------------------------------------------------- async function processar(s) { log(`sessão ${s.id}: iniciando`); if (!s.audio_path) throw new Error('sessão sem audio_path'); const { data: arquivo, error: errDl } = await db.storage.from(BUCKET).download(s.audio_path); if (errDl) throw new Error(`download do áudio: ${errDl.message}`); const nome = s.audio_path.split('/').pop() || 'consulta.webm'; log(`sessão ${s.id}: transcrevendo (${(arquivo.size / 1048576).toFixed(1)} MB)`); const t0 = Date.now(); const { texto, duracao } = await transcrever(arquivo, nome); log(`sessão ${s.id}: transcrito em ${((Date.now() - t0) / 1000).toFixed(0)}s ` + `(${texto.length} caracteres, áudio de ${duracao ? duracao.toFixed(0) : '?'}s)`); if (!texto) throw new Error('transcrição vazia'); const t1 = Date.now(); let resumo = await resumir(texto); log(`sessão ${s.id}: resumido em ${((Date.now() - t1) / 1000).toFixed(0)}s`); // confere os nomes de medicamento contra o catálogo antes de gravar resumo = await conferirMedicamentos(resumo); const { error: errUp } = await db.from('sessoes_gravacao').update({ status: 'concluida', transcricao: texto, resumo, audio_duracao_seg: duracao, modelo_transcricao: WHISPER_MODEL, processado_em: new Date().toISOString(), }).eq('id', s.id); if (errUp) throw new Error(`gravar resultado: ${errUp.message}`); // LGPD: o áudio some assim que vira texto const { error: errRm } = await db.storage.from(BUCKET).remove([s.audio_path]); if (errRm) { log(`sessão ${s.id}: AVISO — falha ao apagar o áudio: ${errRm.message}`); } else { await db.from('sessoes_gravacao') .update({ audio_excluido_em: new Date().toISOString() }) .eq('id', s.id); } log(`sessão ${s.id}: concluída`); } // --------------------------------------------------------------------- // Laço principal // --------------------------------------------------------------------- let rodando = true; process.on('SIGTERM', () => { log('SIGTERM, encerrando após a sessão atual'); rodando = false; }); process.on('SIGINT', () => { log('SIGINT, encerrando após a sessão atual'); rodando = false; }); const dormir = ms => new Promise(r => setTimeout(r, ms)); async function loop() { log(`worker no ar · whisper=${WHISPER_URL} (${WHISPER_MODEL}) · ollama=${OLLAMA_URL} (${OLLAMA_MODEL})`); while (rodando) { let sessao = null; try { const { data, error } = await db.rpc('reivindicar_sessao_gravacao'); if (error) throw new Error(`reivindicar: ${error.message}`); sessao = Array.isArray(data) ? data[0] : data; if (!sessao?.id) { await dormir(Number(INTERVALO_MS)); continue; } await processar(sessao); } catch (e) { log(`ERRO: ${e.message}`); if (sessao?.id) { await db.from('sessoes_gravacao').update({ status: 'erro', erro_mensagem: String(e.message).slice(0, 1000), }).eq('id', sessao.id); } await dormir(5000); } } log('worker encerrado'); process.exit(0); } loop();
